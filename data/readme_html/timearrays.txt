<p dir="auto">TimeArrays simplifies working with time series data. It offers features like basic math operations, sliding window techniques, data resampling, and handling of missing values.</p>

struct OHLC
    open::Float64
    high::Float64
    low::Float64
    close::Float64
end

function ohlc(x::AbstractVector{&lt;:Number})
    return if isempty(x)
        ta_nan(OHLC)
    else
