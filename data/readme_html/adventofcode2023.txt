<div class="highlight highlight-source-julia notranslate position-relative overflow-auto" dir="auto" data-snippet-clipboard-copy-content="(@v1.8) pkg&gt; activate aoc"><pre>(<span class="pl-c1">@v1</span>.<span class="pl-c1">8</span>) pkg<span class="pl-k">&gt;</span> activate aoc</pre></div>

<div class="snippet-clipboard-content notranslate position-relative overflow-auto" data-snippet-clipboard-copy-content="(aoc) pkg&gt; add https://github.com/goggle/AdventOfCode2023.jl"><pre class="notranslate"><code>(aoc) pkg&gt; add https://github.com/goggle/AdventOfCode2023.jl
