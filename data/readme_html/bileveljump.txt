<p dir="auto"><a href="https://github.com/joaquimg/BilevelJuMP.jl">BilevelJuMP.jl</a> is a
<a href="https://github.com/JuMP-dev/JuMP.jl">JuMP</a> extension for
modeling and solving bilevel optimization problems.</p>

<div class="highlight highlight-source-julia notranslate position-relative overflow-auto" dir="auto" data-snippet-clipboard-copy-content="using BilevelJuMP, HiGHS

model = BilevelModel(
    HiGHS.Optimizer,
    mode = BilevelJuMP.FortunyAmatMcCarlMode(primal_big_M = 100, dual_big_M = 100)
)

@variable(Lower(model), x)
@variable(Upper(model), y)

@objective(Upper(model), Min, 3x + y)
@constraints(Upper(model), begin
    x &lt;= 5
    y &lt;= 8
    y &gt;= 0
end)

@objective(Lower(model), Min, -x)
@constraints(Lower(model), begin
     x +  y &lt;= 8
    4x +  y &gt;= 8
    2x +  y &lt;= 13
    2x - 7y &lt;= 0
end)

optimize!(model)

objective_value(model) # = 3 * (3.5 * 8/15) + 8/15 # = 6.13...
value(x) # = 3.5 * 8/15 # = 1.86...
value(y) # = 8/15 # = 0.53..."><pre><span class="pl-k">using</span> BilevelJuMP, HiGHS
