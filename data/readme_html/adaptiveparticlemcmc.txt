<p dir="auto">Simple implementation of adaptive proposals within particle Markov chain Monte Carlo <a href="https://doi.org/10.1111/j.1467-9868.2009.00736.x" rel="nofollow">(Andrieu, Doucet and Holenstein, 2010)</a>, based on the <a href="https://github.com/mvihola/AdaptiveMCMC.jl">AdaptiveMCMC.jl</a> and <a href="https://github.com/awllee/SequentialMonteCarlo.jl">SequentialMonteCarlo.jl</a> packages.</p>
<p dir="auto">The package implemnents the following combinations:</p>
<ul dir="auto">
<li>Particle marginal Metropolis-Hastings with
Adaptive Metropolis covariance adaptation (<a href="https://projecteuclid.org/euclid.bj/1080222083" rel="nofollow">Haario, Saksman and Tamminen, 2001</a>, and <a href="http://dx.doi.org/10.1214/105051606000000286" rel="nofollow">Andrieu and Moulines, 2006</a>)</li>
<li>Particle Gibbs with Robust adaptive Metropolis acceptance rate based
shape adaptation <a href="http://dx.doi.org/10.1007/s11222-011-9269-5" rel="nofollow">(Vihola, 2012)</a> and
Adaptive scaling within Adaptive Metropolis <a href="https://doi.org/10.1007/s11222-008-9110-y" rel="nofollow">(e.g. Andrieu and Thoms,
2008)</a></li>
</ul>
<p dir="auto">These choices are discussed in</p>
<ul dir="auto">
<li>M. Vihola. Ergonomic and reliable Bayesian inference with adaptive Markov chain Monte Carlo. In W. W. Piegrorsch, R. Levine,  H. H. Zhang, and T. C. M. Lee, editors, <em>Handbook of Computational Statistics and Data Science</em>, Wiley, to appear.</li>
</ul>
<p dir="auto">If you use this package in your work, please cite the publication above.</p>

# Define the particle type for the model (here, latent is univariate AR(1))
mutable struct SVParticle
    s::Float64
    SVParticle() = new(0.0) # Void constructor required!
end

# Model parameters:
mutable struct SVParam
    ρ::Float64     # Latent AR(1) coefficient
    σ::Float64     # Latent AR(1) noise sd
    β::Float64     # Latent mean
    σ_s1::Float64  # Latent AR(1) stationary sd
end

# Monthly S&amp;P 500 data (from https://datahub.io/core/s-and-p-500):
data = CSV.read(download(&quot;https://raw.githubusercontent.com/mvihola/AdaptiveParticleMCMC.jl/master/examples/sp500post2000.csv&quot;), DataFrame)
sp500_data = diff(log.(data.SP500)) # Monthly log-returns
sp500_data .-= mean(sp500_data)     # Remove trend
# Initialise parameters
sv_par = SVParam(0.9,1.0,0.0,1.0)

# This will be the SequentialMonteCarlo &quot;particle scratch&quot;, which
# will contain both model data &amp; parameters, and which will be
# the 'scratch' argument of M_ar1!, lM_ar1, lG_sv
struct SVScratch
    par::SVParam
    y::Vector{Float64}
    SVScratch() = new(sv_par, sp500_data)
end

# Transition *simulator* of stationary zero-mean AR(1) with
# parameters (ρ, σ)
function M_ar1!(x, rng, k, x_prev, scratch)
    if k == 1
        x.s = rand(rng, Normal(0.0, scratch.par.σ_s1))
    else
        x.s = rand(rng, Normal(scratch.par.ρ*x_prev.s, scratch.par.σ))
    end
end

# Log transition *density* of stationary zero-mean AR(1) with parameters (ρ, σ)
# (only used with Particle Gibbs, not with Particle marginal Metropolis-Hastings)
function lM_ar1(k, x_prev, x, scratch)
    if k == 1
        return logpdf(Normal(0.0, scratch.par.σ_s1), x.s)
    else
        return logpdf(Normal(scratch.par.ρ*x_prev.s, scratch.par.σ), x.s)
    end
end

# Potential (stochastic volatility observation log-density)
function lG_sv(k, x, scratch)
    s = exp(.5*(scratch.par.β + x.s))        # Observation sd
    logpdf(Normal(0.0, s), scratch.y[k])   # y[k] ~ N(0, s^2)
end

# We are sampling transformed parameters (logit(ρ), log(σ), β)
# The function set_param! sets the model parameters, that is, transforms
# parameters to the 'actual' model parameter values.
inv_logit(x) = 2.0/(1.0+exp(-x)) - 1.0 # (-∞,∞) → (-1,1)
function set_param!(scratch, θ)
    scratch.par.ρ = inv_logit(θ.logit_̢rho)
    scratch.par.σ = exp(θ.log_sigma)
    scratch.par.β = θ.beta
    scratch.par.σ_s1 = scratch.par.σ/sqrt(1.0 - scratch.par.ρ^2) # Stationary variance
end

# Normal prior for the transformed parameters
function prior(theta)
    (logpdf(Normal(6.0,5.0), theta.logit_̢rho)
    +logpdf(Normal(-1.0,5.0), theta.log_sigma)
    +logpdf(Normal(-6.0,5.0), theta.beta))
end

# Create data structures for SequentialMonteCarlo:
##################################################
T = length(sp500_data)
N = 64     # Number of particles
n = 40_000 # Number of PMCMC iterations

# Set up SMC state:
state = SMCState(T, N, SVParticle, SVScratch, set_param!, lG_sv, M_ar1!, lM_ar1)

# Initial (transformed) parameter vector
theta0 = LVector(logit_̢rho=0.0, log_sigma=0.0, beta=0.0)
# Particle marginal Metropolis-Hastings with Adaptive Metropolis
out_pmmh = adaptive_pmmh(theta0, prior, state, n;
  thin=100, show_progress=2, save_paths=true);

# Particle Gibbs with Robust Adaptive Metropolis
out_pg = adaptive_pg(theta0, prior, state, n;
  thin=100, show_progress=2, save_paths=true);

<span class="pl-k">function</span> <span class="pl-en">lM_ar1</span>(k, x_prev, x, scratch)
    <span class="pl-k">if</span> k <span class="pl-k">==</span> <span class="pl-c1">1</span>
        <span class="pl-k">return</span> <span class="pl-c1">logpdf</span>(<span class="pl-c1">Normal</span>(<span class="pl-c1">0.0</span>, scratch<span class="pl-k">.</span>par<span class="pl-k">.</span>σ_s1), x<span class="pl-k">.</span>s)
    <span class="pl-k">else</span>
        <span class="pl-k">return</span> <span class="pl-c1">logpdf</span>(<span class="pl-c1">Normal</span>(scratch<span class="pl-k">.</span>par<span class="pl-k">.</span>ρ<span class="pl-k">*</span>x_prev<span class="pl-k">.</span>s, scratch<span class="pl-k">.</span>par<span class="pl-k">.</span>σ), x<span class="pl-k">.</span>s)
    <span class="pl-k">end</span>
<span class="pl-k">end</span>

<span class="pl-c"><span class="pl-c">#</span> Potential (stochastic volatility observation log-density)</span>
<span class="pl-k">function</span> <span class="pl-en">lG_sv</span>(k, x, scratch)
    s <span class="pl-k">=</span> <span class="pl-c1">exp</span>(.<span class="pl-c1">5</span><span class="pl-k">*</span>(scratch<span class="pl-k">.</span>par<span class="pl-k">.</span>β <span class="pl-k">+</span> x<span class="pl-k">.</span>s))        <span class="pl-c"><span class="pl-c">#</span> Observation sd</span>
    <span class="pl-c1">logpdf</span>(<span class="pl-c1">Normal</span>(<span class="pl-c1">0.0</span>, s), scratch<span class="pl-k">.</span>y[k])   <span class="pl-c"><span class="pl-c">#</span> y[k] ~ N(0, s^2)</span>
<span class="pl-k">end</span>

<span class="pl-c"><span class="pl-c">#</span> We are sampling transformed parameters (logit(ρ), log(σ), β)</span>
<span class="pl-c"><span class="pl-c">#</span> The function set_param! sets the model parameters, that is, transforms</span>
<span class="pl-c"><span class="pl-c">#</span> parameters to the 'actual' model parameter values.</span>
<span class="pl-en">inv_logit</span>(x) <span class="pl-k">=</span> <span class="pl-c1">2.0</span><span class="pl-k">/</span>(<span class="pl-c1">1.0</span><span class="pl-k">+</span><span class="pl-c1">exp</span>(<span class="pl-k">-</span>x)) <span class="pl-k">-</span> <span class="pl-c1">1.0</span> <span class="pl-c"><span class="pl-c">#</span> (-∞,∞) → (-1,1)</span>
<span class="pl-k">function</span> <span class="pl-en">set_param!</span>(scratch, θ)
    scratch<span class="pl-k">.</span>par<span class="pl-k">.</span>ρ <span class="pl-k">=</span> <span class="pl-c1">inv_logit</span>(θ<span class="pl-k">.</span>logit_̢rho)
    scratch<span class="pl-k">.</span>par<span class="pl-k">.</span>σ <span class="pl-k">=</span> <span class="pl-c1">exp</span>(θ<span class="pl-k">.</span>log_sigma)
    scratch<span class="pl-k">.</span>par<span class="pl-k">.</span>β <span class="pl-k">=</span> θ<span class="pl-k">.</span>beta
    scratch<span class="pl-k">.</span>par<span class="pl-k">.</span>σ_s1 <span class="pl-k">=</span> scratch<span class="pl-k">.</span>par<span class="pl-k">.</span>σ<span class="pl-k">/</span><span class="pl-c1">sqrt</span>(<span class="pl-c1">1.0</span> <span class="pl-k">-</span> scratch<span class="pl-k">.</span>par<span class="pl-k">.</span>ρ<span class="pl-k">^</span><span class="pl-c1">2</span>) <span class="pl-c"><span class="pl-c">#</span> Stationary variance</span>
<span class="pl-k">end</span>

<span class="pl-c"><span class="pl-c">#</span> Normal prior for the transformed parameters</span>
<span class="pl-k">function</span> <span class="pl-en">prior</span>(theta)
    (<span class="pl-c1">logpdf</span>(<span class="pl-c1">Normal</span>(<span class="pl-c1">6.0</span>,<span class="pl-c1">5.0</span>), theta<span class="pl-k">.</span>logit_̢rho)
    <span class="pl-k">+</span><span class="pl-c1">logpdf</span>(<span class="pl-c1">Normal</span>(<span class="pl-k">-</span><span class="pl-c1">1.0</span>,<span class="pl-c1">5.0</span>), theta<span class="pl-k">.</span>log_sigma)
    <span class="pl-k">+</span><span class="pl-c1">logpdf</span>(<span class="pl-c1">Normal</span>(<span class="pl-k">-</span><span class="pl-c1">6.0</span>,<span class="pl-c1">5.0</span>), theta<span class="pl-k">.</span>beta))
<span class="pl-k">end</span>
