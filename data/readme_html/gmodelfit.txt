unc  = [1.1, 1.1, 1.1, 1.2, 1.2]
dom  = Domain(x)
data = Measures(dom, meas, unc)

# initial guess values:

# Fit model to the data
bestfit, stats = fit(model, data)"><pre><span class="pl-k">using</span> GModelFit
