<p dir="auto"><code>QuasiEwald.jl</code> is a package written in <code>Julia</code>.
It is an implementation of the algorithm Quasi Ewald Method, which used to calculate the electrostatic interaction in dielectric confined Quasi-2D charged systems for MD simulations, which has a linear complexity to the number of particles.</p>
<div class="markdown-heading" dir="auto"><h2 class="heading-element" dir="auto">Getting Started</h2><a id="user-content-getting-started" class="anchor" aria-label="Permalink: Getting Started" href="#getting-started"><svg class="octicon octicon-link" viewBox="0 0 16 16" version="1.1" width="16" height="16" aria-hidden="true"><path d="m7.775 3.275 1.25-1.25a3.5 3.5 0 1 1 4.95 4.95l-2.5 2.5a3.5 3.5 0 0 1-4.95 0 .751.751 0 0 1 .018-1.042.751.751 0 0 1 1.042-.018 1.998 1.998 0 0 0 2.83 0l2.5-2.5a2.002 2.002 0 0 0-2.83-2.83l-1.25 1.25a.751.751 0 0 1-1.042-.018.751.751 0 0 1-.018-1.042Zm-4.69 9.64a1.998 1.998 0 0 0 2.83 0l1.25-1.25a.751.751 0 0 1 1.042.018.751.751 0 0 1 .018 1.042l-1.25 1.25a3.5 3.5 0 1 1-4.95-4.95l2.5-2.5a3.5 3.5 0 0 1 4.95 0 .751.751 0 0 1-.018 1.042.751.751 0 0 1-1.042.018 1.998 1.998 0 0 0-2.83 0l-2.5 2.5a1.998 1.998 0 0 0 0 2.83Z"></path></svg></a></div>
<p dir="auto">This package has to be used as an extentation of the author's previous package <a href="https://github.com/ArrogantGao/ExTinyMD.jl"><code>ExTinyMD.jl</code></a>, which is a small but fast MD package written in <code>Julia</code> language.
To use this package, you only need to type</p>
<div class="highlight highlight-source-julia notranslate position-relative overflow-auto" dir="auto" data-snippet-clipboard-copy-content="pkg&gt; add ExTinyMD, QuasiEwald"><pre>pkg<span class="pl-k">&gt;</span> add ExTinyMD, QuasiEwald</pre></div>
<p dir="auto">in your command lines.</p>

begin
    n_atoms = 2
    L = 180.0
    boundary = ExTinyMD.Q2dBoundary(L, L, 10.0) 

    atoms = [Atom(type = 1, mass = 1.0, charge = 1.0), Atom(type = 2, mass = 1.0, charge = -1.0)]

    sys = MDSys(
                n_atoms = n_atoms,
                atoms = atoms,
                boundary = boundary,
                interactions = [(NoInteraction(), NoNeighborFinder(n_atoms))],
                loggers = [TrajectionLogger(step = 100, output = false)],
                simulator = VerletProcess(dt = 0.001, thermostat = AndersenThermoStat(1.0, 0.05))
            )

    Force_x = Vector{Vector{Float64}}()
    coord_1 = Point(50.0, 50.0, 1.0)
    X = 0.1:0.1:40.0

    for (γ_1, γ_2) in [(0.0, 0.0), (0.95, 0.95), (-0.95, -0.95), (10.0, 10.0), (-10.0, -10.0)]
        ϵ_0 = 1.0
        n_t = 30

        accuracy = 1e-4
        α = 1.0
        r_c = 15.0
        k_c = sqrt(-4 * α * log(accuracy))

        force_x = Vector{Float64}()

        for x_2 in X
            info = SimulationInfo(n_atoms, atoms, (0.0, L, 0.0, L, 0.0, 10.0), boundary; min_r = 1.0, temp = 1.0)
            coord_2 = Point(50.0 + x_2, 50.0, 1.01)

            info.particle_info[2].position = coord_2

            sortz = SortingFinder(info)
            cellq2d = CellListQ2D(info, r_c, boundary, 1)
            interaction_short = QuasiEwaldShortInteraction(γ_1, γ_2, ϵ_0, (L, L, 10.0), false, accuracy, α, n_atoms, r_c, n_t)
            interaction_long = QuasiEwaldLongInteraction(γ_1, γ_2, ϵ_0, (L, L, 10.0), false, accuracy, α, n_atoms, k_c, 0)

            QuasiEwald_Fs!(interaction_short, cellq2d, sys, info)
            QuasiEwald_Fl!(interaction_long, sortz, sys, info)
            push!(force_x, info.particle_info[1].acceleration[1])
        end
        push!(Force_x, force_x)
    end

    plot(dpi = 300, size = (800, 600), legend = :topright, xlabel = &quot;x&quot;, ylabel = &quot;force_x&quot;)
    for (γ, force_x) in zip([0.0, 0.95, -0.95, 10.0, -10.0], Force_x)
        plot!(X, force_x, label = &quot;γ = &quot; * string(γ), ylim = [-0.06, 0.06])
    end
    savefig(&quot;force_x.png&quot;)

    n_atoms = 436

    L_x = 100.0
    L_y = 100.0

    for i in 1:218

    end

        push!(atoms, Atom(type = 2, mass = 1.0, charge = - 1.0))
    end

    (γ_1, γ_2) = (0.95, -0.95)

    info = SimulationInfo(n_atoms, atoms, (0.0, L_x, 0.0, L_y, 0.5, L_z - 0.5), boundary; min_r = 2.0, temp = 1.0)

    ϵ_0 = 1.0

    accuracy = 1e-4
    α = 1.0
    k_c = sqrt(- 4 * α * log(accuracy))
    r_c = (α * accuracy)^(-1/3) / 2
    n_t = 30
    rbe_p = 50

    intershort = QuasiEwaldShortInteraction(γ_1, γ_2, ϵ_0, L, true, accuracy, α, n_atoms, r_c, n_t)
    short_finder = CellListQ2D(info, r_c + 1.0, boundary, 100)
    interlong = QuasiEwaldLongInteraction(γ_1, γ_2, ϵ_0, L, true, accuracy, α, n_atoms, k_c, rbe_p)
    long_finder = SortingFinder(info)

    interactions = [
        (LennardJones(), CellList3D(info, 4.5, boundary, 100)),
        (SubLennardJones(0.0, L_z; cutoff = 0.5, σ = 0.5), SubNeighborFinder(1.0, info, 0.0, L_z)), 
        (intershort, short_finder),
        (interlong, long_finder)
        ]

    loggers = [TempartureLogger(100, output = true), TrajectionLogger(step = 100, output = true)]
    simulator = VerletProcess(dt = 0.001, thermostat = AndersenThermoStat(1.0, 0.05))

    sys = MDSys(

        interactions = interactions,

        (<span class="pl-c1">LennardJones</span>(), <span class="pl-c1">CellList3D</span>(info, <span class="pl-c1">4.5</span>, boundary, <span class="pl-c1">100</span>)),
        (<span class="pl-c1">SubLennardJones</span>(<span class="pl-c1">0.0</span>, L_z; cutoff <span class="pl-k">=</span> <span class="pl-c1">0.5</span>, σ <span class="pl-k">=</span> <span class="pl-c1">0.5</span>), <span class="pl-c1">SubNeighborFinder</span>(<span class="pl-c1">1.0</span>, info, <span class="pl-c1">0.0</span>, L_z)), 
        (intershort, short_finder),
        (interlong, long_finder)
        ]

    loggers <span class="pl-k">=</span> [<span class="pl-c1">TempartureLogger</span>(<span class="pl-c1">100</span>, output <span class="pl-k">=</span> <span class="pl-c1">true</span>), <span class="pl-c1">TrajectionLogger</span>(step <span class="pl-k">=</span> <span class="pl-c1">100</span>, output <span class="pl-k">=</span> <span class="pl-c1">true</span>)]
    simulator <span class="pl-k">=</span> <span class="pl-c1">VerletProcess</span>(dt <span class="pl-k">=</span> <span class="pl-c1">0.001</span>, thermostat <span class="pl-k">=</span> <span class="pl-c1">AndersenThermoStat</span>(<span class="pl-c1">1.0</span>, <span class="pl-c1">0.05</span>))

    sys <span class="pl-k">=</span> <span class="pl-c1">MDSys</span>(
        n_atoms <span class="pl-k">=</span> n_atoms,
        atoms <span class="pl-k">=</span> atoms,
        boundary <span class="pl-k">=</span> boundary,
        interactions <span class="pl-k">=</span> interactions,
        loggers <span class="pl-k">=</span> loggers,
        simulator <span class="pl-k">=</span> simulator
    )

    <span class="pl-c1">simulate!</span>(simulator, sys, info, <span class="pl-c1">100000</span>)
<span class="pl-k">end</span></pre></div>
