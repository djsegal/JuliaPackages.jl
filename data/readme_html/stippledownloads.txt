<p dir="auto">StippleDownloads is a plugin for Stipple to enable download of dynamically generated files.
The event-based handlers guarantee that only the requesting client receives a copy of the file.</p>
<p dir="auto">There is support for text and binary files, filenames can be freely chosen.</p>

<p dir="auto"><strong>Downloading a variable's content as raw data or as BSON</strong></p>
<div class="highlight highlight-source-julia notranslate position-relative overflow-auto" dir="auto" data-snippet-clipboard-copy-content="using Stipple, Stipple.ReactiveTools
using StippleUI, StippleDownloads
using BSON

@app begin
    @in data = randn(1000)

    @event download_raw begin
        download_binary(__model__, data, &quot;raw&quot;)
    end
    @event download_bson begin
        # calling @save on `data` throws an error because it is a reactive variable. Need to make a copy
        data_var = data
        io = IOBuffer()
        BSON.@save io data_var
        seekstart(io)
        download_binary(__model__, take!(io), &quot;data.bson&quot;)
    end

end

function ui()
    row([
            cell(btn(class=&quot;q-ml-lg&quot;, &quot;Raw&quot;, icon=&quot;download&quot;, @on(:click, :download_raw, :addclient), color=&quot;primary&quot;, nocaps=true))
            cell(btn(class=&quot;q-ml-lg&quot;, &quot;BSON&quot;, icon=&quot;download&quot;, @on(:click, :download_bson, :addclient), color=&quot;secondary&quot;, nocaps=true))
        ])
    
end

<p dir="auto"><strong>Downloading a DataFrame as an Excel sheet</strong></p>
<div class="highlight highlight-source-julia notranslate position-relative overflow-auto" dir="auto" data-snippet-clipboard-copy-content="using Stipple, Stipple.ReactiveTools
using StippleUI
using StippleDownloads

using DataFrames
using XLSX

import Stipple.opts
import StippleUI.Tables.table

function df_to_xlsx(df)
    io = IOBuffer()
    XLSX.writetable(io, df)
    take!(io)
end

@app begin
    @out table = DataTable(DataFrame(:a =&gt; rand(1:10, 5), :b =&gt; rand(1:10, 5)))
    @in text = &quot;The quick brown fox jumped over the ...&quot;

    @event download_text begin
        download_text(__model__, :text)
    end

    @event download_df begin
        try
            download_binary(__model__, df_to_xlsx(table.data), &quot;file.xlsx&quot;; client = event[&quot;_client&quot;])
        catch ex
            println(ex)
        end
    end
end

function ui()
    row(cell(class = &quot;st-module&quot;, [

        row([
            cell(textfield(class = &quot;q-pr-md&quot;, &quot;Download text&quot;, :text, placeholder = &quot;no output yet ...&quot;, :outlined, :filled, type = &quot;textarea&quot;))
            cell(table(class = &quot;q-pl-md&quot;, :table))
        ])
              
        row([
            cell(col = 1, &quot;Without client info&quot;)
            cell(btn(&quot;Text File&quot;, icon = &quot;download&quot;, @on(:click, :download_text), color = &quot;primary&quot;, nocaps = true))
            cell(col = 1, &quot;With client info&quot;)
            cell(btn(class = &quot;q-ml-lg&quot;, &quot;Excel File&quot;, icon = &quot;download&quot;, @on(:click, :download_df, :addclient), color = &quot;primary&quot;, nocaps = true))
        ])
    ]))
end

@page(&quot;/&quot;, ui)

<p dir="auto">To see the difference between calling with or without client info, duplicate the applications's tab and click the 'Download Text' button.</p>
<p dir="auto">Two identical files will be downloaded, because duplicating the tab establishes a synchronised copy of your app. To ensure that only the requesting client receives a file, you should restrict the download to the client via <code>event["_client"]</code>
and make sure that the triggering button has an additional <code>:addclient</code> to include this information.</p>
