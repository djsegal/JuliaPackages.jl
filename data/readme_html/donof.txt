mol = """
0 1
 O  0.0000   0.000   0.121
 H  0.0000   0.751  -0.485
 H  0.0000  -0.751  -0.485
"""

bset,p = DoNOF.molecule(mol,"cc-pvtz")

p.ipnof = 8

p.RI = true
p.gpu = false

p.orb_method = "Rotations"

E,C,gamma,fmiug0 = DoNOF.energy(bset,p)
</code></pre></div>
