<p dir="auto">Ever wanted to read Configurations.jl from environment variables? Here you go 🙂</p>
<div class="highlight highlight-source-julia notranslate position-relative overflow-auto" dir="auto" data-snippet-clipboard-copy-content="using Configurations
@option struct Opt1
    a::Bool
    b::Union{Nothing, Int} = nothing
end
@option struct Opt2
    a::Opt1
    b::String

using ConfigurationsENV

# without an explicit `env` argument to `from_env`, the global `ENV` is used
env = Dict(
    &quot;PREFIX_A_A&quot; =&gt; &quot;true&quot;,
    &quot;PREFIX_A_B&quot; =&gt; &quot;&quot;,
    &quot;PREFIX_B&quot; =&gt; &quot;hello world&quot;,
)

config = from_env(Opt2, env, prefix=&quot;PREFIX_&quot;, separator=&quot;_&quot;)

<p dir="auto">In case you just want to parse a subset of the options fields from environment variables, use <code>return_dict=true</code>.</p>
